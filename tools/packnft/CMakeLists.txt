cmake_minimum_required(VERSION 3.16)
project(packnft C)
set(PACKNFT_SOURCES catalog.c reveal.c metadata.c msg.c txn.c core_ix.c)
add_library(packnft_lib STATIC ${PACKNFT_SOURCES})
target_include_directories(packnft_lib PUBLIC . ../../shared)
target_compile_options(packnft_lib PRIVATE -std=c11 -O0 -g)
add_executable(packnft cli.c)
target_link_libraries(packnft PRIVATE packnft_lib)
target_compile_options(packnft PRIVATE -std=c11 -O0 -g)
add_executable(packnft_test test.c)
target_link_libraries(packnft_test PRIVATE packnft_lib)
target_compile_options(packnft_test PRIVATE -std=c11 -O0 -g)
add_executable(test_msg test_msg.c)
target_link_libraries(test_msg PRIVATE packnft_lib)
add_executable(test_core test_core.c)
target_link_libraries(test_core PRIVATE packnft_lib)
